200                 val progress = progressPercent / 100.0
201                 sendProgress(progress, "processing", "Export en cours $progressPercent%")
202                 if (progressPolling) {
203                     progressHandler?.postDelayed(this, 200)
204                 }
205             }
206         }
207         progressHandler = Handler(Looper.getMainLooper())
208         progressHandler?.postDelayed(poll, 200)
209     }

211     private fun parseComposition(json: String): Triple<List<ClipItem>, List<ClipItem>, List<TextItem>> {
212         val root = JSONObject(json)
213         val texts = mutableListOf<TextItem>()
214         val ta = root.optJSONArray("texts")
215         if (ta != null) {
216             for (i in 0 until ta.length()) {
217                 val o = ta.getJSONObject(i)
218                 texts.add(TextItem(
219                     o.optString("text", ""), o.optLong("startMs", 0L), o.optLong("durationMs", 0L),
220                     o.optLong("color", 0xFFFFFFFFL).toInt(), o.optDouble("size", 56.0), o.optDouble("y", 0.5)
221                 ))
222             }
223         }
224         return Triple(parseArray(root.optJSONArray("video")), parseArray(root.optJSONArray("audio")), texts)
225     }

227     private fun parseArray(arr: JSONArray?): List<ClipItem> {
228         val list = mutableListOf<ClipItem>()
229         if (arr == null) return list
230         for (i in 0 until arr.length()) {
231             val o = arr.getJSONObject(i)
232             list.add(ClipItem(
233                 o.getString("path"), o.optLong("startMs", 0L), o.optLong("durationMs", 0L),
234                 o.optDouble("speed", 1.0), o.optDouble("volume", 1.0),
235                 o.optDouble("brightness", 0.0), o.optDouble("contrast", 0.0),
236                 o.optDouble("saturation", 0.0), o.optDouble("sepia", 0.0)
237             ))
238         }
239         return list
240     }

242     /** Applique le rognage (start/duration), la resolution et l'option audio. */
243     private fun buildItem(item: ClipItem, removeAudio: Boolean, height: Int): EditedMediaItem {
244         val uri = if (item.path.contains("://")) Uri.parse(item.path) else Uri.fromFile(File(item.path))
245         val mediaItem = MediaItem.Builder().setUri(uri).apply {
246             if (item.durationMs > 0) {
247                 setClippingConfiguration(
248                     MediaItem.ClippingConfiguration.Builder()
249                         .setStartPositionMs(item.startMs)
250                         .setEndPositionMs(item.startMs + item.durationMs)
251                         .build()
252                 )
253             }
254         }.build()
255         val b = EditedMediaItem.Builder(mediaItem).setRemoveAudio(removeAudio)
256         val audioProcs = mutableListOf<AudioProcessor>()
257         val videoEffects = mutableListOf<Effect>()
258         if (item.speed != 1.0) {
259             audioProcs.add(SonicAudioProcessor().apply { setSpeed(item.speed.toFloat()) })
260             videoEffects.add(SpeedChangeEffect(item.speed.toFloat()))
261         }
262         if (item.volume != 1.0) {
263             val mixer = ChannelMixingAudioProcessor()
264             for (ch in 1..2) {
265                 mixer.putChannelMixingMatrix(ChannelMixingMatrix.create(ch, ch).scaleBy(item.volume.toFloat()))
266             }
267             audioProcs.add(mixer)
268         }
269         // Filtres : meme ordre que l'apercu (saturation, sepia, contraste, luminosite).
270         if (item.saturation != 0.0) {
271             videoEffects.add(HslAdjustment.Builder().adjustSaturation((item.saturation * 100).toFloat()).build())
272         }
273         if (item.sepia > 0.0) videoEffects.add(SepiaMatrix(item.sepia.toFloat()))
274         if (item.contrast != 0.0) videoEffects.add(Contrast(item.contrast.toFloat()))
275         if (item.brightness != 0.0) videoEffects.add(Brightness(item.brightness.toFloat()))
276         if (height > 0) videoEffects.add(Presentation.createForHeight(height))
277         if (audioProcs.isNotEmpty() || videoEffects.isNotEmpty()) {
278             b.setEffects(Effects(audioProcs, videoEffects))
279         }
280         return b.build()
281     }
